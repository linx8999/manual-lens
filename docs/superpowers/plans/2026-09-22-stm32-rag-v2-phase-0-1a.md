# STM32 RAG V2 Phase 0 + 1A Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the first evidence-driven visual retrieval loop so text found only inside a PNG/JPG or scanned PDF becomes searchable evidence, opens the correct original, and highlights the OCR bounding box without requiring chat or embedding APIs.

**Architecture:** Preserve the Electron/TypeScript/SQLite application and add a versioned evidence schema beside the legacy chunk schema. A local CPython 3.11 RapidOCR sidecar communicates with the Electron main process over versioned JSONL on stdio; imported evidence is staged by document revision, indexed in FTS5, atomically activated, resolved through strict IPC, and rendered with normalized bbox overlays.

**Tech Stack:** Electron 44, React 19, TypeScript 5.9, Node 24 `node:sqlite`, SQLite FTS5, Zod 4, Vitest 5, Playwright 1.63, CPython 3.11 x64, RapidOCR 3.9.2, ONNX Runtime 1.26.0 CPU, pypdfium2 5.13.0, Pillow, uv 0.11+, PowerShell.

**Spec:** `docs/superpowers/specs/2026-09-22-stm32-rag-evidence-harness-v2-design.md`

## Global Constraints

- Target Windows x64 CPU; Phase 1A uses the project-local Python 3.11 sidecar and never launches a shell to invoke it.
- OCR defaults to RapidOCR's Simplified Chinese model, which also recognizes English and ASCII STM32 identifiers.
- Sidecar stdout contains JSONL protocol only; diagnostics go to stderr.
- Every sidecar message contains `schemaVersion`, `jobId`, and `requestId`; duplicate request IDs are idempotent.
- OCR images and scanned PDF pages are processed one page at a time at 300 DPI; decoded bitmaps are released after each page.
- Bboxes use unrotated source media, top-left origin, normalized `[x, y, width, height]` values in `[0, 1]`.
- New citations store immutable `segmentId + revision`; legacy citations remain readable and never guess a replacement segment.
- New retrieval reads only `documents.active_revision`; staging and superseded revisions are invisible.
- OCR/parser failure preserves the original document and reports partial or failed coverage; it does not silently claim full searchability.
- No runtime model download, remote OCR, arbitrary Renderer file path, or deletion of legacy chunk tables.
- Phase 1A acceptance runs with chat and embedding APIs disabled.
- Existing uncommitted user changes are preserved. No execution starts until the user explicitly authorizes the local baseline snapshot commit described below.

export function clampNormalizedBbox(value: readonly number[]): NormalizedBbox {
  if (value.length !== 4 || value.some((item) => !Number.isFinite(item))) {
    throw new Error("Invalid normalized bbox.");
  }
  const x = Math.min(1, Math.max(0, value[0]!));
  const y = Math.min(1, Math.max(0, value[1]!));
  const right = Math.min(1, Math.max(x, x + value[2]!));
  const bottom = Math.min(1, Math.max(y, y + value[3]!));
  return [x, y, right - x, bottom - y];
}

export function rotateNormalizedBbox(bbox: NormalizedBbox, rotation: QuarterRotation): NormalizedBbox {
  const [x, y, width, height] = bbox;
  if (rotation === 90) return [1 - y - height, x, height, width];
  if (rotation === 180) return [1 - x - width, 1 - y - height, width, height];
  if (rotation === 270) return [y, 1 - x - width, height, width];
  return bbox;
}

export function pageNumberOf(location: EvidenceLocation): number | null {
  return location.kind === "pdf_page" ? location.pageNumber : null;
}

- [ ] **Step 3: Implement ordered checksum-protected migrations**

```ts
export interface Migration {
  version: number;
  checksum: string;
  up(database: DatabaseSync): void;
}

export interface BackupResult {
  databasePath: string;
  manifestPath: string;
  sha256: string;
  integrity: "ok";
}

export function runMigrations(database: DatabaseSync, migrations: Migration[]): void {
  database.exec(`CREATE TABLE IF NOT EXISTS schema_migrations (
    version INTEGER PRIMARY KEY,
    applied_at INTEGER NOT NULL,
    checksum TEXT NOT NULL
  )`);
  for (const migration of [...migrations].sort((a, b) => a.version - b.version)) {
    const existing = database.prepare("SELECT checksum FROM schema_migrations WHERE version = ?").get(migration.version) as { checksum: string } | undefined;
    if (existing) {
      if (existing.checksum !== migration.checksum) throw new Error(`Migration ${migration.version} checksum mismatch.`);
      continue;
    }
    database.exec("BEGIN IMMEDIATE");
    try {
      migration.up(database);
      database.prepare("INSERT INTO schema_migrations(version, applied_at, checksum) VALUES (?, ?, ?)").run(migration.version, Date.now(), migration.checksum);
      database.prepare("INSERT OR REPLACE INTO schema_meta(key, value) VALUES ('schema_version', ?)").run(String(migration.version));
      database.exec("COMMIT");
    } catch (error) {
      database.exec("ROLLBACK");
      throw error;
    }
  }
}
```

- [ ] **Step 4: Implement a verified SQLite backup**

```ts
import { backup, DatabaseSync } from "node:sqlite";
import { createHash } from "node:crypto";
import { mkdir, readFile, writeFile } from "node:fs/promises";
import { join } from "node:path";

export async function createVerifiedBackup(source: DatabaseSync, backupDir: string): Promise<BackupResult> {
  await mkdir(backupDir, { recursive: true });
  const databasePath = join(backupDir, `knowledge-${Date.now()}.db`);
  await backup(source, databasePath);
  const bytes = await readFile(databasePath);
  const sha256 = createHash("sha256").update(bytes).digest("hex");
  const verification = new DatabaseSync(databasePath, { readOnly: true });
  const integrity = String((verification.prepare("PRAGMA integrity_check").get() as { integrity_check: string }).integrity_check);
  verification.close();
  if (integrity !== "ok") throw new Error(`Backup integrity check failed: ${integrity}`);
  const manifestPath = `${databasePath}.json`;
  await writeFile(manifestPath, JSON.stringify({ databasePath, sha256, integrity }, null, 2), "utf8");
  return { databasePath, manifestPath, sha256, integrity: "ok" };
}
```

export interface OcrParser {
  parse(request: OcrParseRequest, signal?: AbortSignal): Promise<OcrPageResult>;
  cancel(jobId: string): Promise<void>;
  close(): Promise<void>;
}

interface PendingRequest {
  promise: Promise<OcrPageResult>;
  resolve(result: OcrPageResult): void;
  reject(error: Error): void;
  timer: ReturnType<typeof setTimeout>;
  jobId: string;
}

export class OcrSidecarClient implements OcrParser {
  private readonly pending = new Map<string, PendingRequest>();
  private readonly completed = new Map<string, OcrPageResult>();

  async parse(request: OcrParseRequest, signal?: AbortSignal): Promise<OcrPageResult> {
    const cached = this.completed.get(request.requestId);
    if (cached) return cached;
    const pending = this.pending.get(request.requestId);
    if (pending) return pending.promise;
    const created = this.createPending(request, signal);
    this.pending.set(request.requestId, created);
    this.write(request);
    return created.promise;
  }
}

`createPending()` constructs the shown internal record; its timeout clears the map entry before rejecting.

```python
from rapidocr import RapidOCR

class OcrEngine:
    def __init__(self) -> None:
        self._engine = RapidOCR()

    def recognize(self, image) -> list[dict]:
        output = self._engine(image)
        if output.boxes is None or output.txts is None or output.scores is None:
            return []
        width, height = image.size
        segments = []
        for points, text, score in zip(output.boxes, output.txts, output.scores, strict=True):
            if not text or not text.strip():
                continue
            xs = [float(point[0]) for point in points]
            ys = [float(point[1]) for point in points]
            left, right = min(xs), max(xs)
            top, bottom = min(ys), max(ys)
            bbox = [left / width, top / height, (right - left) / width, (bottom - top) / height]
            segments.append({"text": text.strip(), "bbox": [max(0.0, min(1.0, value)) for value in bbox], "confidence": float(score)})
        return segments
```

```ts
export function useDragPan(options: { ignoreTarget?: (target: Element) => boolean } = {}) {
  const session = useRef<{
    pointerId: number;
    startX: number;
    startY: number;
    startScrollLeft: number;
    startScrollTop: number;
    dragging: boolean;
  } | null>(null);
  const [dragging, setDragging] = useState(false);

  const onPointerDown = (event: React.PointerEvent<HTMLElement>): void => {
    if (event.button !== 0 || options.ignoreTarget?.(event.target as Element)) return;
    session.current = {
      pointerId: event.pointerId,
      startX: event.clientX,
      startY: event.clientY,
      startScrollLeft: event.currentTarget.scrollLeft,
      startScrollTop: event.currentTarget.scrollTop,
      dragging: false
    };
    event.currentTarget.setPointerCapture?.(event.pointerId);
  };

  const onPointerMove = (event: React.PointerEvent<HTMLElement>): void => {
    const active = session.current;
    if (!active || active.pointerId !== event.pointerId) return;
    const dx = event.clientX - active.startX;
    const dy = event.clientY - active.startY;
    if (!active.dragging && Math.hypot(dx, dy) < 4) return;
    active.dragging = true;
    setDragging(true);
    event.preventDefault();
    event.currentTarget.scrollLeft = active.startScrollLeft - dx;
    event.currentTarget.scrollTop = active.startScrollTop - dy;
  };

  const finish = (event: React.PointerEvent<HTMLElement>): void => {
    if (session.current?.pointerId !== event.pointerId) return;
    event.currentTarget.releasePointerCapture?.(event.pointerId);
    session.current = null;
    setDragging(false);
  };

  return {
    dragging,
    handlers: { onPointerDown, onPointerMove, onPointerUp: finish, onPointerCancel: finish }
  };
}

CSS uses `cursor: grab`, `[data-panning="true"] { cursor: grabbing; user-select: none; }`, while `.textLayer span { cursor: text; }`. async function expectGrabPan(page: Page, viewport: Locator): Promise<void> {
  await expect(viewport).toBeVisible();
  await viewport.evaluate((element) => {
    element.scrollLeft = 40;
    element.scrollTop = 60;
  });
  const before = await viewport.evaluate((element) => ({ left: element.scrollLeft, top: element.scrollTop }));
  const box = await viewport.boundingBox();
  if (!box) throw new Error("Reader viewport has no bounding box.");
  await page.mouse.move(box.x + box.width * 0.65, box.y + box.height * 0.65);
  await page.mouse.down();
  await page.mouse.move(box.x + box.width * 0.65 - 120, box.y + box.height * 0.65 - 90, { steps: 6 });
  await expect(viewport).toHaveAttribute("data-panning", "true");
  await page.mouse.up();
  await expect(viewport).toHaveAttribute("data-panning", "false");
  await expect.poll(() => viewport.evaluate((element) => element.scrollLeft)).toBeGreaterThan(before.left);
  await expect.poll(() => viewport.evaluate((element) => element.scrollTop)).toBeGreaterThan(before.top);
}
```
